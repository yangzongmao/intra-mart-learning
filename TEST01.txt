package com.example.csvupdate.diff;

/**
 * 表示一个字段的变化
 */
public class ColumnDiff {

    private final String column;
    private final Object before;
    private final Object after;

    public ColumnDiff(String column, Object before, Object after) {
        this.column = column;
        this.before = before;
        this.after = after;
    }

    public String getColumn() { return column; }
    public Object getBefore() { return before; }
    public Object getAfter() { return after; }

    public boolean isChanged() {
        return true;
    }
}

package com.example.csvupdate.diff;

import java.util.ArrayList;
import java.util.List;

/**
 * 一行数据的差异
 */
public class RowDiff {

    private final String pk;
    private final List<ColumnDiff> changes = new ArrayList<>();

    private boolean notFound;

    public RowDiff(String pk) {
        this.pk = pk;
    }

    public static RowDiff notFound(String pk) {
        RowDiff r = new RowDiff(pk);
        r.notFound = true;
        return r;
    }

    public void addChange(String column, Object before, Object after) {
        changes.add(new ColumnDiff(column, before, after));
    }

    public boolean hasChanges() {
        return !changes.isEmpty();
    }

    public String getPk() { return pk; }
    public List<ColumnDiff> getChanges() { return changes; }
    public boolean isNotFound() { return notFound; }
}

/**
 * Diff统计结果（用于画面显示）
 */
public class DiffSummary {

    private int total;
    private int changed;
    private int unchanged;
    private int notFound;

    public void incTotal() { total++; }
    public void incChanged() { changed++; }
    public void incUnchanged() { unchanged++; }
    public void incNotFound() { notFound++; }

    // getter省略
}

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * 构造统一主键字符串
 */
public class PkBuilder {

    private static final String SEP = "|";

    public static String build(
            Map<String, Object> row,
            List<String> pkColumns) {

        return pkColumns.stream()
                .map(col -> {
                    Object v = row.get(col);
                    return v == null ? "" : v.toString();
                })
                .collect(Collectors.joining(SEP));
    }
}

package com.example.csvupdate.diff;

/**
 * 统一值比较规则
 */
public class ValueNormalizer {

    public static String normalize(Object v) {

        if (v == null) return null;

        String s = v.toString().trim();

        // CSV规则：NULL => null
        if ("NULL".equalsIgnoreCase(s))
            return null;

        // 空字符串允许
        if (s.isEmpty())
            return "";

        return s;
    }

    public static boolean equals(Object v1, Object v2) {
        return java.util.Objects.equals(
                normalize(v1),
                normalize(v2)
        );
    }
}

package com.example.csvupdate.diff;

import java.util.*;
import java.util.function.Function;
import java.util.stream.Collectors;

/**
 * 通用Diff引擎
 *
 * 支持：
 *  - 任意表
 *  - 多主键
 *  - CSV更新规则
 *  - 高性能O(N)
 */
public class DiffEngine {

    /**
     * 执行Diff
     */
    public DiffResult diff(
            List<Map<String,Object>> csvRows,
            List<Map<String,Object>> dbRows,
            DiffContext context) {

        // ========= 1. 构建DB索引 =========

        Map<String, Map<String,Object>> dbIndex =
                dbRows.stream()
                        .collect(Collectors.toMap(
                                r -> PkBuilder.build(
                                        r,
                                        context.getPkColumns()),
                                Function.identity()
                        ));

        List<RowDiff> diffs = new ArrayList<>();
        DiffSummary summary = new DiffSummary();

        // ========= 2. 逐行比较 =========

        for (Map<String,Object> csvRow : csvRows) {

            summary.incTotal();

            String pk = PkBuilder.build(
                    csvRow,
                    context.getPkColumns());

            Map<String,Object> dbRow = dbIndex.get(pk);

            if (dbRow == null) {
                diffs.add(RowDiff.notFound(pk));
                summary.incNotFound();
                continue;
            }

            RowDiff rowDiff = new RowDiff(pk);

            // ========= 字段比较 =========
            for (String col : context.getCompareColumns()) {

                Object oldVal = dbRow.get(col);
                Object newVal = csvRow.get(col);

                if (!ValueNormalizer.equals(oldVal, newVal)) {

                    rowDiff.addChange(col, oldVal, newVal);
                }
            }

            if (rowDiff.hasChanges()) {
                diffs.add(rowDiff);
                summary.incChanged();
            } else {
                summary.incUnchanged();
            }
        }

        return new DiffResult(diffs, summary);
    }
}

public class GenericSqlProvider {

    public String bulkUpsert(Map<String, Object> param) {

        String tableName = (String) param.get("tableName");
        List<String> columns =
                (List<String>) param.get("columns");
        List<String> pkColumns =
                (List<String>) param.get("pkColumns");
        List<Map<String, Object>> dataList =
                (List<Map<String, Object>>) param.get("dataList");

        StringBuilder sql = new StringBuilder();

        // INSERT INTO table (...)
        sql.append("INSERT INTO ")
           .append(tableName)
           .append(" (");

        for (int i = 0; i < columns.size(); i++) {
            sql.append(columns.get(i));
            if (i < columns.size() - 1) {
                sql.append(",");
            }
        }

        sql.append(") VALUES ");

        // VALUES (...)
        for (int i = 0; i < dataList.size(); i++) {

            sql.append("(");

            for (int j = 0; j < columns.size(); j++) {

                sql.append("#{dataList[")
                   .append(i)
                   .append("].")
                   .append(columns.get(j))
                   .append("}");

                if (j < columns.size() - 1) {
                    sql.append(",");
                }
            }

            sql.append(")");

            if (i < dataList.size() - 1) {
                sql.append(",");
            }
        }

        // ON DUPLICATE KEY UPDATE
        sql.append(" ON DUPLICATE KEY UPDATE ");

        boolean first = true;

        for (String col : columns) {

            // 主键字段不更新
            if (pkColumns.contains(col)) {
                continue;
            }

            if (!first) {
                sql.append(",");
            }

            sql.append(col)
               .append("=VALUES(")
               .append(col)
               .append(")");

            first = false;
        }

        return sql.toString();
    }
}
